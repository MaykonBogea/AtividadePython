inicio
    retorne (celsius * 9 / 5) + 32
fimfuncao


funcao logico validar_senha(cadeia senha)
inicio

funcao real caixa(real precos[], inteiro quantidade)
var
    real total
    inteiro i
inicio
    total <- 0

    para i de 0 ate quantidade - 1 faca
        total <- total + precos[i]
    fimpara

    retorne total
fimfuncao
